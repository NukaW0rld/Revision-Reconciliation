    Since these are alert-related, add to a new separate path prefix or inline in the runs router.
    Use a simple approach: add the dismiss route to the runs router with path `/alerts/dismiss/{alert_id}`.

    ```python
    @router.post("/alerts/dismiss/{alert_id}", response_class=HTMLResponse)
    def dismiss_alert(
        alert_id: int,
        request: Request,
        db: Session = Depends(get_db),
        user: User = Depends(get_current_user),
    ):
        from shop.models import RunAlert
        alert = db.query(RunAlert).filter(RunAlert.id == alert_id).first()
        if alert and alert.user_id == user.id:
            alert.is_read = True
            db.commit()
        # Return empty response — HTMX will swap the banner div out
        return HTMLResponse("")

    @router.post("/{run_id}/acknowledge-warning")
    def acknowledge_warning(
        run_id: int,
        request: Request,
        db: Session = Depends(get_db),
        user: User = Depends(get_current_user),
    ):
        # Phase 3 will handle the actual review queue
        # For now, redirect to a placeholder
    ```
