    ```python
    import os
    import logging
    from huey import SqliteHuey

    logger = logging.getLogger(__name__)

    HUEY_DB = os.environ.get("HUEY_DB", "/app/data/huey.db")
    OUT_DIR = os.environ.get("OUT_DIR", "/app/out")

    huey = SqliteHuey(filename=HUEY_DB)

    STAGE_NAMES = [
        "Form 3 parsing",
        "Balloon detection",
        "Text extraction",
        "Anchor building",
        "Alignment",
        "Candidate matching",
        "Classification",
        "Output",
    ]

    def _update_stage(db, run, stage_idx: int, stage_name: str):
        run.current_stage_index = stage_idx
        run.current_stage = stage_name
        db.commit()

    def _create_alert(db, run):
        from shop.models import RunAlert
        if run.reviewer_id is None:
            return
        alert = RunAlert(
            run_id=run.id,
            user_id=run.reviewer_id,
            message=f"Run {run.id} ({run.part_number}) failed at stage: {run.failure_stage or 'Unknown'}",
            failure_stage=run.failure_stage,
            is_read=False,
        )
        db.add(alert)
        db.commit()

    @huey.task()
    def run_pipeline_task(
        run_id: int,
        revA_path: str,
        revB_path: str,
        form3_path: str,
        part_name: str,
        dpi: int = 300,
    ):
        """
        Huey task: calls delta_preservation.cli.run_pipeline() with stage_callback.
        Updates Run row at each stage; handles failure and warning states.
        """
        from shop.database import SessionLocal
        from shop.models import Run
        from delta_preservation.cli import run_pipeline
        from delta_preservation.config import MIN_ALIGNMENT_RATIO

        db = SessionLocal()
        try:
            run = db.query(Run).filter(Run.id == run_id).first()
            if run is None:
                logger.error("run_pipeline_task: Run %d not found", run_id)
                return

            run.status = "running"
            run.current_stage_index = 0
            run.current_stage = STAGE_NAMES[0]
            db.commit()

            # Track state needed for balloon and alignment checks
            _balloon_count = {"value": None}
            _alignment_ratio = {"value": None}
            _alignment_inliers = {"value": None}

            def stage_callback(stage_idx: int, stage_name: str):
                _update_stage(db, run, stage_idx, stage_name)
                db.refresh(run)

            out_dir = run_pipeline(
                revA_pdf=revA_path,
                revB_pdf=revB_path,
                form3_xlsx=form3_path,
                out_dir=OUT_DIR,
                dpi=dpi,
                part_name=part_name,
                stage_callback=stage_callback,
            )

            # After pipeline completes, check alignment confidence from output
            # The pipeline writes delta_packet.json — read it to check confidence stats
            import json
            packet_path = out_dir / "delta_packet.json"
            if packet_path.exists():
                with open(packet_path) as f:
                    packet = json.load(f)
                items = packet.get("items", [])
                if items:
                    scores = [item.get("scores", {}).get("location", 1.0) for item in items]
                    low_conf = sum(1 for s in scores if s < 0.5)
                    ratio = low_conf / len(scores)
                    if ratio > 0.5:  # majority of characteristics have low location confidence
                        run.status = "warning"
                        run.warning_type = "low_confidence"
                        run.confidence_summary = {
                            "low_confidence_count": low_conf,
                            "total": len(scores),
                            "low_confidence_ratio": round(ratio, 2),
                            "message": f"{int(ratio*100)}% of characteristics have low location confidence in Rev B",
                        }
                        db.commit()
                        run.output_dir = str(out_dir)
                        db.commit()
                        return

            run.status = "completed"
            run.output_dir = str(out_dir)
            db.commit()

        except Exception as exc:
            logger.exception("run_pipeline_task failed for run %d", run_id)
            db.refresh(run)
            run.status = "failed"
            run.failure_stage = run.current_stage or "Unknown"
            run.failure_message = str(exc)
            db.commit()
            _create_alert(db, run)
        finally:
            db.close()
    ```

    Note on Rev A balloon failure (PIPE-04): The pipeline calls `detect_balloons()` in Stage 2 and returns an empty dict if no balloons found. However, the pipeline does not raise an exception for empty balloons — it just produces empty anchors and a nearly-empty delta packet. To hard-fail on empty Rev A balloons, we need a check after Stage 2 completes. Since stage_callback is called BEFORE each stage, this approach won't intercept stage 2 results in-flight.

If `len(items) == 0`, set status=failed, failure_stage="Balloon detection", failure_message="No characteristics found in Rev A PDF. Ensure the drawing uses numbered balloon annotations."
