# FastAPI native SSE (0.135.x)
from fastapi.sse import EventSourceResponse, ServerSentEvent
from collections.abc import AsyncIterable
import asyncio

@router.get("/{run_id}/sse")
async def run_progress_sse(
    run_id: int,
    request: Request,
    db: Session = Depends(get_db),
    user: User = Depends(get_current_user),
) -> AsyncIterable[ServerSentEvent]:
    terminal = {"completed", "failed", "warning"}
    while True:
        if await request.is_disconnected():
            break
        run = db.query(Run).filter(Run.id == run_id).first()
        data = {
            "status": run.status,
            "current_stage_index": run.current_stage_index,
            "current_stage": run.current_stage,
            "failure_stage": run.failure_stage,
            "failure_message": run.failure_message,
            "warning_type": run.warning_type,
        }
        yield ServerSentEvent(data=json.dumps(data), event="stage_update")
        if run.status in terminal:
            yield ServerSentEvent(event="close", data="done")
            break
        await asyncio.sleep(1.0)
```

    import json as _json

    STAGE_NAMES = [
        "Form 3 parsing", "Balloon detection", "Text extraction",
        "Anchor building", "Alignment", "Candidate matching", "Classification", "Output"

    @router.get("/{run_id}", response_class=HTMLResponse)
    def run_status(
        run_id: int,
        request: Request,
        db: Session = Depends(get_db),
        user: User = Depends(get_current_user),
    ):
        run = db.query(Run).filter(Run.id == run_id).first()
        if run is None:
            raise HTTPException(status_code=404, detail="Run not found")
        unread_count = db.query(RunAlert).filter(
            RunAlert.user_id == user.id, RunAlert.is_read == False
        ).count()
        from shop.models import ShopConfig
        config = db.query(ShopConfig).filter(ShopConfig.id == 1).first()
        shop_name = config.shop_name if config else "Delta Preservation"
        stages = [{"index": i, "name": n} for i, n in enumerate(STAGE_NAMES)]
        return templates.TemplateResponse(request, "runs/status.html", {
            "user": user, "run": run, "stages": stages,
            "unread_alert_count": unread_count, "shop_name": shop_name,
        })

    @router.get("/{run_id}/sse", response_class=EventSourceResponse)
    async def run_sse(
        run_id: int,
        request: Request,
        db: Session = Depends(get_db),
        user: User = Depends(get_current_user),
    ) -> AsyncIterable[ServerSentEvent]:
        terminal = {"completed", "failed", "warning"}
        while True:
            if await request.is_disconnected():
                break
            run = db.query(Run).filter(Run.id == run_id).first()
            if run is None:
                break
            payload = _json.dumps({
                "status": run.status,
                "current_stage_index": run.current_stage_index,
                "current_stage": run.current_stage,
                "failure_stage": run.failure_stage,
                "failure_message": run.failure_message,
                "warning_type": run.warning_type,
                "confidence_summary": run.confidence_summary,
            })
            yield ServerSentEvent(data=payload, event="stage_update")
            if run.status in terminal:
                yield ServerSentEvent(event="close", data="done")
                break
            await asyncio.sleep(1.0)

    @router.post("/{run_id}/abort")
    def abort_run(
        run_id: int,
        request: Request,
        db: Session = Depends(get_db),
        user: User = Depends(get_current_user),
    ):
        run = db.query(Run).filter(Run.id == run_id).first()
        if run is None:
            raise HTTPException(status_code=404, detail="Run not found")
        if run.status == "warning" and run.warning_type == "low_confidence":
            run.status = "failed"
            run.failure_stage = "Alignment"
            run.failure_message = "Run aborted by engineer due to low alignment confidence."
            db.commit()
        return RedirectResponse(f"/runs/{run_id}", status_code=302)
    ```

    Create shop/templates/runs/_stage_checklist.html:
    This is the SSE swap target. It must work both as the initial render and as the SSE-swapped partial.
    Since SSE sends JSON, use a simple approach: the _stage_checklist.html is rendered server-side on page load and replaced by HTMX on each SSE event. However, HTMX SSE swaps the innerHTML of the target — so for initial render, just include this partial in status.html. For SSE updates, the server needs to send HTML (not JSON) OR the page needs JavaScript to update from JSON.

    Architecture decision (Claude's discretion): Use JS to update the checklist from the JSON SSE payload rather than a server-rendered partial. This avoids needing a server-side HTMX partial endpoint for each SSE event.

    Revised approach:
    - The SSE event sends JSON payload
    - A small inline `<script>` in status.html listens to the SSE event via native EventSource (not HTMX) and updates the DOM from JSON
    - The stage checklist is a static DaisyUI steps list that is updated by JS based on `current_stage_index` and `status`

    This avoids the complexity of HTMX partial responses from SSE (which would require the server to render HTML in the SSE event data, not JSON).

    - SSE listener script (inline JS using native EventSource, not HTMX):
      ```html
      <script>
      {% if run.status not in ["completed", "failed", "warning"] %}
      const evtSource = new EventSource("/runs/{{ run.id }}/sse");
      evtSource.addEventListener("stage_update", function(e) {
          const data = JSON.parse(e.data);
          updateChecklist(data);
      });
      evtSource.addEventListener("close", function(e) {
          evtSource.close();
          // Reload page to show terminal state (warning/failed/completed UI)
          window.location.reload();
      });

      function updateChecklist(data) {
          const stageNames = ["Form 3 parsing", "Balloon detection", "Text extraction",
              "Anchor building", "Alignment", "Candidate matching", "Classification", "Output"];
          const ul = document.getElementById("stage-checklist");
          if (!ul) return;
          ul.innerHTML = stageNames.map((name, i) => {
              const isDone = i < data.current_stage_index || data.status === "completed";
              const isCurrent = i === data.current_stage_index && data.status === "running";
              const isFailed = data.failure_stage === name && data.status === "failed";
              let cls = "step";
              if (isDone) cls += " step-success";
              else if (isCurrent) cls += " step-primary";
              else if (isFailed) cls += " step-error";
              let inner = name;
              if (isDone) inner = `<span>&#10003; ${name}</span>`;
              else if (isCurrent) inner = `<span class="flex items-center gap-1"><span class="loading loading-spinner loading-xs"></span>${name}</span>`;
              else if (isFailed) inner = `<span class="font-bold text-error">&#10007; ${name}</span>`;
              return `<li class="${cls}">${inner}</li>`;
          }).join("");
      }
      {% endif %}
      </script>
      ```
