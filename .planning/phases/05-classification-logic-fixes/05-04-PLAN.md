```python
def reconcile_removed_added_pairs(
    items: List[DeltaItem],
    anchors: List[Anchor],
) -> List[DeltaItem]:
    """Collapse same-page, spatially-close, type-compatible removed+added
    pairs into single `changed` items.

    Contract:
      - Input: the full classified list (anchor items, then added items).
      - Output: a new list with paired removeds rewritten as `changed` in
        place and their matched added items dropped. List order is preserved
        except for the removed added items.
      - Each added item pairs with at most one removed item.
      - Each removed item pairs with at most one added item (closest wins).
      - Proximity is Euclidean centroid distance in PDF points.
    """
    if not items:
        return list(items)

    removed_items = [i for i in items if i.status == "removed"]
    added_items = [
        i for i in items
        if i.status == "added" and i.added_span is not None
    ]
    if not removed_items or not added_items:
        return list(items)

    anchor_by_char = {a.char_no: a for a in anchors}
    used_added_char_nos: Set[int] = set()

    for removed in removed_items:
        anchor = anchor_by_char.get(removed.char_no)
        if anchor is None:
            continue
        anchor_bbox = anchor.req_bbox or anchor.balloon_bbox
        if anchor_bbox is None:
            continue
        ax0, ay0, ax1, ay1 = anchor_bbox
        anchor_cx = (ax0 + ax1) / 2.0
        anchor_cy = (ay0 + ay1) / 2.0
        anchor_type = classify_requirement_type(anchor.requirement_raw)

        best_added: Optional[DeltaItem] = None
        best_distance: float = float("inf")

        for added in added_items:
            if added.char_no in used_added_char_nos:
                continue
            if added.added_span is None:
                continue
            bx0, by0, bx1, by1 = added.added_span.bbox_pdf
            added_cx = (bx0 + bx1) / 2.0
            added_cy = (by0 + by1) / 2.0
            distance = math.sqrt(
                (anchor_cx - added_cx) ** 2 + (anchor_cy - added_cy) ** 2
            )
            if distance > CLS02_MAX_DISTANCE_PT:
                continue
            added_type = classify_requirement_type(added.added_span.text)
            if are_requirement_types_incompatible(anchor_type, added_type):
                continue
            if distance < best_distance:
                best_distance = distance
                best_added = added

        if best_added is None:
            continue

        used_added_char_nos.add(best_added.char_no)
        strict = best_distance <= CLS02_STRICT_DISTANCE_PT
        removed.status = "changed"
        removed.confidence = 0.75 if strict else 0.60
        rev_b_snippet = (best_added.added_span.text or "").strip()[:80]
        removed.reasons = [
            f"Removed + added pair reconciled as changed: Rev A anchor and "
            f"Rev B unmatched span within {best_distance:.0f} pt on same "
            f"page (threshold {CLS02_MAX_DISTANCE_PT:.0f} pt)",
            f'Rev B text: "{rev_b_snippet}"',
            *list(best_added.reasons),
        ]
        removed.added_span = best_added.added_span
        removed.match = None

    return [
        item for item in items
        if not (item.status == "added" and item.char_no in used_added_char_nos)
    ]
```
