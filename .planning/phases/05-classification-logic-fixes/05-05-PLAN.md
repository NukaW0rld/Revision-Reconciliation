"""Phase 5 regression harness.

Purpose:
  Guard against the Phase 5 classification-logic helpers over-firing on
  previously-`conforming` items captured in the 9-part debug-report
  corpus snapshots. Full-pipeline regeneration against ground truth is
  Phase 7's job (VER-01); this file operates on the pre-serialized
  JSON snapshots and exercises the new helpers directly.

Scope:
  - Reads only `assets/debug_report_part*.json` (snapshots).
  - NEVER reads or writes `assets/part*/ground_truth.json`.
  - NEVER invokes `run_pipeline` or PyMuPDF.

The whitelist thresholds below are intentionally loose. If Phase 7
regenerates the snapshots with the fixes applied, the expected
helper-True counts may shift; re-tightening is Phase 7's concern.
"""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any, Dict, Iterator, List, Tuple

import pytest

from delta_preservation.reconcile.classify import (
    CLS02_MAX_DISTANCE_PT,
    CLS02_STRICT_DISTANCE_PT,
    _ASYMMETRIC_SHAPE_RE,
    _looks_like_adjacency_bleed,
    reconcile_removed_added_pairs,
)

_REPO_ROOT = Path(__file__).resolve().parent.parent
_SNAPSHOT_DIR = _REPO_ROOT / "assets"

# Whitelist caps: how many conforming+unchanged items are allowed to
# trigger each helper across the 9 snapshots. Set with margin above
# the known exemplars to allow structural siblings without forcing
# per-part identity checks.
_MAX_BLEED_HITS = 8
_MAX_ASYMMETRIC_HITS = 4


def _snapshot_paths() -> List[Path]:
    return sorted(_SNAPSHOT_DIR.glob("debug_report_part*.json"))


def _iter_items() -> Iterator[Tuple[str, Dict[str, Any]]]:
    """Yield (snapshot_name, item_dict) for every item across all parts."""
    for path in _snapshot_paths():
        payload = json.loads(path.read_text())
        items = payload.get("items") or payload.get("delta_items") or []
        for item in items:
            if isinstance(item, dict):
                yield path.name, item


def _evaluation_status(item: Dict[str, Any]) -> str:
    eval_obj = item.get("evaluation") or {}
    status = eval_obj.get("status") if isinstance(eval_obj, dict) else None
    return status or ""


# ---------------------------------------------------------------------------
# Infrastructure sanity
# ---------------------------------------------------------------------------


def test_snapshot_directory_is_populated():
    paths = _snapshot_paths()
    assert len(paths) >= 1, (
        f"Expected debug_report_part*.json under {_SNAPSHOT_DIR}; found none"
    )


# ---------------------------------------------------------------------------
# Baseline snapshot: helpers must not over-fire on previously-conforming items
# ---------------------------------------------------------------------------


class TestPhase5BaselineSnapshot:
    """Verify the new Phase-5 helpers do not over-fire on items that
    already classified as conforming+unchanged in the snapshot corpus."""

    def test_bleed_helper_does_not_overfire_on_conforming_unchanged_items(self):
        hits: List[Tuple[str, int, str]] = []
        for snap_name, item in _iter_items():
            if _evaluation_status(item) != "conforming":
                continue
            if item.get("pipeline_classification") != "unchanged":
                continue
            rev_b = item.get("requirement_revB") or ""
            if _looks_like_adjacency_bleed(rev_b):
                hits.append((snap_name, item.get("char_no", -1), rev_b))
        assert len(hits) <= _MAX_BLEED_HITS, (
            f"_looks_like_adjacency_bleed over-fires on "
            f"{len(hits)} conforming+unchanged items (max {_MAX_BLEED_HITS}): "
            f"{hits}"
        )

    def test_asymmetric_shape_does_not_overfire_on_conforming_unchanged_items(self):
        hits: List[Tuple[str, int, str]] = []
        for snap_name, item in _iter_items():
            if _evaluation_status(item) != "conforming":
                continue
            if item.get("pipeline_classification") != "unchanged":
                continue
            rev_b = item.get("requirement_revB") or ""
            if _ASYMMETRIC_SHAPE_RE.search(rev_b):
                hits.append((snap_name, item.get("char_no", -1), rev_b))
        assert len(hits) <= _MAX_ASYMMETRIC_HITS, (
            f"_ASYMMETRIC_SHAPE_RE over-fires on "
            f"{len(hits)} conforming+unchanged items "
            f"(max {_MAX_ASYMMETRIC_HITS}): {hits}"
        )

    def test_known_exemplars_still_trigger_helpers(self):
        """Positive gate: confirm the Phase-5 helpers fire on the
        canonical bleed + asymmetric exemplars so the fix is live."""
        bleed_positive = _looks_like_adjacency_bleed(
            "4 x Ø8 THRU ALL / ⌴ Ø13.5 ↧ 8.5"
        )
        bleed_positive_p4 = _looks_like_adjacency_bleed(
            "2X Ø.201 ↧ 0.50 / 1/4-20 UNC - 2B"
        )
        asym_positive = bool(
            _ASYMMETRIC_SHAPE_RE.search("2X 22.0° +0.3° / −0.1°")
        )
        assert bleed_positive, (
            "Part-1 char-11 canonical bleed span not detected by "
            "_looks_like_adjacency_bleed"
        )
        assert bleed_positive_p4, (
            "Part-4 char-7 canonical bleed span not detected by "
            "_looks_like_adjacency_bleed"
        )
        assert asym_positive, (
            "Part-7 char-4 canonical asymmetric span not matched by "
            "_ASYMMETRIC_SHAPE_RE"
        )


# ---------------------------------------------------------------------------
# Helper generalization: smoke and total-input robustness
# ---------------------------------------------------------------------------


class TestPhase5HelpersGeneralize:
    """Sanity/robustness coverage for the three Phase-5 helpers against
    every item's requirement text across the corpus."""

    def test_reconcile_function_is_importable_and_accepts_empty(self):
        assert reconcile_removed_added_pairs([], []) == []
        assert CLS02_MAX_DISTANCE_PT == 150.0
        assert CLS02_STRICT_DISTANCE_PT == 75.0

    def test_bleed_helper_handles_every_requirement_without_exception(self):
        checked = 0
        for _snap_name, item in _iter_items():
            for key in ("requirement_revA", "requirement_revB"):
                value = item.get(key) or ""
                # Must return bool without raising on any real corpus text.
                result = _looks_like_adjacency_bleed(value)
                assert isinstance(result, bool)
                checked += 1
        assert checked > 0, "No requirement strings were exercised"

    def test_asymmetric_shape_handles_every_requirement_without_exception(self):
        checked = 0
        for _snap_name, item in _iter_items():
            for key in ("requirement_revA", "requirement_revB"):
                value = item.get(key) or ""
                # .search must not raise on any real corpus text.
                match = _ASYMMETRIC_SHAPE_RE.search(value)
                assert match is None or match is not None  # type sanity
                checked += 1
        assert checked > 0, "No requirement strings were exercised"
