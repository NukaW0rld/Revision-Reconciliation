```python
# ---------------------------------------------------------------------------
# Wave 0: confidence_flags field — backward-compat deserialization
# ---------------------------------------------------------------------------

class TestConfidenceFlagsBackwardCompat:
    """Adding `confidence_flags` to the Pydantic DeltaItem must not break
    legacy packet deserialization (see 05-RESEARCH.md Assumption A4,
    Pitfall 7)."""

    def test_pydantic_default_is_empty_list_and_not_aliased(self):
        from delta_preservation.types import DeltaItem as PydanticDeltaItem
        a = PydanticDeltaItem(
            char_no=1, status="unchanged", confidence=0.5,
            reasons=["r"], scores={},
        )
        b = PydanticDeltaItem(
            char_no=2, status="unchanged", confidence=0.5,
            reasons=["r"], scores={},
        )
        assert a.confidence_flags == []
        assert b.confidence_flags == []
        a.confidence_flags.append("x")
        assert b.confidence_flags == [], (
            "confidence_flags default is aliased across instances"
        )

    def test_legacy_json_without_confidence_flags_deserializes(self):
        """A JSON payload missing `confidence_flags` must validate to
        a DeltaItem with `confidence_flags == []`."""
        from delta_preservation.types import DeltaItem as PydanticDeltaItem
        legacy_payload = {
            "char_no": 1,
            "status": "unchanged",
            "confidence": 0.9,
            "reasons": ["legacy packet"],
            "scores": {"location": 0.9, "text": 1.0, "context": 0.5},
        }
        model = PydanticDeltaItem.model_validate(legacy_payload)
        assert model.confidence_flags == []

    def test_existing_debug_report_packets_deserialize(self):
        """Load each `assets/debug_report_part*.json` and validate every
        delta_items entry as a Pydantic DeltaItem."""
        import json
        from pathlib import Path
        from delta_preservation.types import DeltaItem as PydanticDeltaItem

        report_dir = Path(__file__).resolve().parent.parent / "assets"
        report_paths = sorted(report_dir.glob("debug_report_part*.json"))
        assert report_paths, "No debug_report_part*.json found under assets/"
        validated_any = False
        for path in report_paths:
            payload = json.loads(path.read_text())
            items = payload.get("delta_items") or payload.get("items") or []
            for item in items:
                if not isinstance(item, dict) or "status" not in item:
                    continue
                PydanticDeltaItem.model_validate(item)
                validated_any = True
        assert validated_any, (
            "No delta_items found across debug_report_part*.json — check "
            "payload shape or report schema."
        )
```

Do not alter any other Pydantic field. Do not add Optional[List[str]]; the field is `List[str]` with a default factory.
