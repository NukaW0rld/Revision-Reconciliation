```python
# --- CLS-01: adjacency-bleed detection --------------------------------------
# See .planning/phases/05-classification-logic-fixes/05-RESEARCH.md Pattern 1.

_BLEED_SLASH_RE = re.compile(r"\s*/\s*")
_COUNT_PREFIX_RE = re.compile(r"\b\d+\s*[Xx](?!\s*\d)")
_GDT_BLEED_SYMBOLS = frozenset({
    "⌖", "⌓", "⟂", "⏥", "∠", "∥", "⊙", "⌒", "⌴", "⌿", "○",
})
_TOLERANCE_CHUNK_RE = re.compile(r"^[+\u2212\-]\s*\d+(?:\.\d+)?\s*°?$")
_FIT_DESIGNATOR_RE = re.compile(r"^[A-Za-z]\d+$")
_BARE_DIGITS_RE = re.compile(r"^\d+$")


def _looks_like_adjacency_bleed(span_text: str) -> bool:
    """Return True when the Rev B span text shows multi-balloon merge markers.

    Heuristic (all must hold):
      (a) splitting on whitespace-padded `/` yields >= 2 non-empty chunks,
      (b) chunks are NOT all bare digits (rejects fractions like '70 / 30'),
      (c) chunks are NOT all fit designators (rejects 'H7/p6'),
      (d) NO chunk is a tolerance continuation (rejects '+0.3 / -0.1',
          preserving CLS-03's signal),
      AND at least one of:
      (i)   >= 2 distinct count prefixes across chunks, OR
      (ii)  a chunk starts with a GD&T control symbol (bleed into a plain
            dimension balloon), OR
      (iii) one chunk has a count prefix and another chunk has an engineering
            symbol (Ø/R/⌴/↧) without a count prefix.
    """
    if not span_text or "/" not in span_text:
        return False
    chunks = [c.strip() for c in _BLEED_SLASH_RE.split(span_text) if c.strip()]
    if len(chunks) < 2:
        return False

    # Reject fit designators like 'H7/p6'
    if all(_FIT_DESIGNATOR_RE.match(c) for c in chunks):
        return False
    # Reject bare-digit fractions like '70 / 30'
    if all(_BARE_DIGITS_RE.match(c) for c in chunks):
        return False
    # Reject tolerance continuations like '+0.3 / -0.1' or '+0.3° / −0.1°'
    if all(_TOLERANCE_CHUNK_RE.match(c) for c in chunks):
        return False

    count_prefix_chunks = [c for c in chunks if _COUNT_PREFIX_RE.search(c)]

    # Signal (i): multiple count prefixes across chunks
    if len(count_prefix_chunks) >= 2:
        return True
    # Signal (ii): a chunk starts with a GD&T control symbol
    if any(c and c[0] in _GDT_BLEED_SYMBOLS for c in chunks):
        return True
    # Signal (iii): one chunk has a count prefix, another has an engineering
    # symbol without a count prefix (e.g. '4 x Ø8 THRU ALL' / '⌴ Ø13.5 ↧ 8.5')
    if count_prefix_chunks:
        engineering_symbols = ("Ø", "R", "⌴", "↧", "⌀", "∅")
        for c in chunks:
            if _COUNT_PREFIX_RE.search(c):
                continue
            if any(sym in c for sym in engineering_symbols):
                return True
    return False
```
