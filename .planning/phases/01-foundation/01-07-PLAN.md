Create `docker/Dockerfile` with three stages:

```dockerfile
# Stage 1: Tailwind CSS compilation (Node.js)
FROM node:22-alpine AS tailwind-builder
WORKDIR /app
COPY package.json ./
RUN npm install
COPY static/src/ ./static/src/
# Copy all templates so Tailwind can scan for class usage
COPY shop/templates/ ./shop/templates/
RUN npm run build:css

# Stage 2: Python dependency builder using uv
FROM python:3.11-slim AS python-builder
COPY --from=ghcr.io/astral-sh/uv:latest /uv /uvx /bin/
WORKDIR /app
COPY pyproject.toml uv.lock ./
# Install dependencies without installing the project itself
RUN uv sync --locked --no-install-project

# Stage 3: Runtime
FROM python:3.11-slim AS runtime
WORKDIR /app
# Copy virtualenv from builder
COPY --from=python-builder /app/.venv /app/.venv
# Copy compiled CSS from Tailwind builder
COPY --from=tailwind-builder /app/static/dist/output.css ./static/dist/output.css
# Copy application source
COPY delta_preservation/ ./delta_preservation/
COPY shop/ ./shop/
COPY static/js/ ./static/js/
COPY static/src/ ./static/src/
COPY run.py run_web.py ./
# Activate venv
ENV PATH="/app/.venv/bin:$PATH"
# Create data directories
RUN mkdir -p /app/data /app/out
# Expose port
EXPOSE 8000
CMD ["python", "run_web.py"]
```

Key constraints:
- Use `python:3.11-slim` — NOT 3.12 or 3.13; pyproject.toml requires `>=3.10,<3.13`
- `uv sync --locked` ensures exact lockfile versions (DEPLOY-02)
- Node.js is only in the tailwind-builder stage (not in runtime) — DEPLOY-03 compliant
- No `npm install` or `pip install` in runtime stage — all bundled in builder

Create `run_web.py` at project root — startup entrypoint:

```python
"""
Startup entrypoint for Docker.
1. Seeds the default admin user from env vars if not present.
2. Starts uvicorn.
"""
import os
import logging
import subprocess
import sys

logging.basicConfig(level=logging.INFO, format="%(levelname)s %(message)s")
logger = logging.getLogger(__name__)

def seed_admin():
    """Create default admin user if no users exist."""
    admin_email = os.environ.get("ADMIN_EMAIL", "admin@shop.local")
    admin_password = os.environ.get("ADMIN_DEFAULT_PASSWORD", "changeme")
    # Override DATABASE_URL if set in environment
    db_url = os.environ.get("DATABASE_URL", "sqlite:///./shop.db")
    # Update database URL in app config before import
    os.environ.setdefault("DATABASE_URL", db_url)
    from sqlalchemy import create_engine
    from sqlalchemy.orm import sessionmaker
    from shop.database import Base
    from shop.models import User, ShopConfig
    from shop.services.auth import hash_password
    engine = create_engine(db_url, connect_args={"check_same_thread": False})
    Base.metadata.create_all(engine)
    Session = sessionmaker(bind=engine)
    with Session() as db:
        existing = db.query(User).filter(User.role == "admin").first()
        if not existing:
            admin = User(
                email=admin_email,
                hashed_password=hash_password(admin_password),
                role="admin",
                is_active=True,
            )
            db.add(admin)
            db.commit()
            logger.info(f"Seeded default admin: {admin_email}")
        else:
            logger.info(f"Admin already exists: {existing.email}")
        # Ensure singleton ShopConfig row exists
        config = db.query(ShopConfig).filter(ShopConfig.id == 1).first()
        if not config:
            db.add(ShopConfig(id=1))
            db.commit()
            logger.info("Created ShopConfig singleton")

if __name__ == "__main__":
    seed_admin()
    logger.info("Starting uvicorn on 0.0.0.0:8000")
    subprocess.run(
        ["uvicorn", "shop.app:app", "--host", "0.0.0.0", "--port", "8000"],
        check=True,
    )
```
